\subsection{Lie Algebra Computations}
\label{subsec:lie-algebra-computations}

In this section, we recite a well-known Lie-algebraic method for proving approximate universality of unitary gadgets in a purely computational manner. Here is a high level overview of the approach: Using some Lie Algebra tricks, it can be shown that given a gate set, it is universal if and only if it generates the Lie algebra of the special unitary group $SU(2^n)$. It turns out that we get the operators $-iZ_1, \ldots, -iZ_n$ for ``free'' given our gadget definition. We have three ways of generating new operators from the above: conjugation by the Clifford circuit attached to the gadget, addition by Trotterization, and commutation. All three of these operations can be done efficiently via the tableau representation in Definition \ref{def:tableau}, with respect to the number of qubits of the gadget. Therefore, we can compute the Lie algebra generated by the gate set, and if it is the full Lie algebra, then the gate set is universal.

Note that for simplicity, we ignore global phases throughout. In other words, we work inside $SU(2^n)$ rather than $U(2^n)$.

\begin{definition}[Clifford group]
\label{def:clifford-group}
The \term{$n$-qubit Clifford group} is the normalizer of the Pauli group in the unitary group:
\begin{align}
    \cC_n
    :=
    \{ C \in U(2^n) \mid C \cP_n C^\dagger = \cP_n \}.
\end{align}
Equivalently, $C \in \cC_n$ if and only if conjugation by $C$ sends Pauli operators to Pauli operators.
\end{definition}

\begin{definition}[Matrix exponential]
\label{def:matrix-exponential}
For any $A \in \C^{N \times N}$, define the \term{matrix exponential} of $A$ by
\begin{align}
    e^A
    :=
    \sum_{k=0}^\infty \frac{A^k}{k!}.
\end{align}
\end{definition}

\begin{definition}
\label{def:hermitian-skew-traceless}
Let $A \in \C^{n \times n}$. We say that $A$ is \term{Hermitian} if $A^\dagger=A$, \term{skew-Hermitian} if $A^\dagger=-A$, and \term{traceless} if $\tr(A)=0$, where \term{$\tr(A)$} is the \term{trace} of $A$, defined by
\begin{align}
    \tr(A)
    =
    \sum_{j=1}^n A_{jj}.
\end{align}
\end{definition}

\begin{definition}
\label{def:su-lie-algebra}
The \term{Lie algebra of $SU(2^n)$} is
\begin{align}
    \mathfrak{su}(2^n)
    :=
    \{ A \in \C^{2^n \times 2^n}
    \mid
    A^\dagger = -A,\;
    \tr(A)=0
    \}.
\end{align}
Equivalently,
\begin{align}
    \mathfrak{su}(2^n)
    =
    \{ -iH \mid H^\dagger = H,\; \tr(H)=0 \}.
\end{align}
Thus $\mathfrak{su}(2^n)$ consists exactly of the traceless skew-Hermitian operators.
\end{definition}

\begin{fact}
\label{fact:exponential-su}
For every $U \in SU(2^n)$, there exists $A \in \mathfrak{su}(2^n)$ such that
\begin{align}
    U = e^A.
\end{align}
\end{fact}

\begin{proof}
Since $U$ is unitary, the spectral theorem gives
\begin{align}
    U
    =
    V \Diag(e^{i\theta_1},\ldots,e^{i\theta_N}) V^\dagger,
\end{align}
where $N=2^n$ and $\theta_j \in \R$ for all $j \in [N]$. Since $\det(U)=1$, we have
\begin{align}
    1
    =
    e^{i(\theta_1+\cdots+\theta_N)}.
\end{align}
Therefore $\theta_1+\cdots+\theta_N \in 2\pi \Z$. By changing one $\theta_j$ by an integer multiple of $2\pi$, we may assume
\begin{align}
    \sum_{j=1}^N \theta_j = 0.
\end{align}
Now define
\begin{align}
    A
    :=
    V \Diag(i\theta_1,\ldots,i\theta_N) V^\dagger.
\end{align}
Then $A^\dagger=-A$, $\tr(A)=i\sum_j \theta_j=0$, so $A \in \mathfrak{su}(N)$, and
\begin{align}
    e^A
    =
    V \Diag(e^{i\theta_1},\ldots,e^{i\theta_N}) V^\dagger
    =
    U.
\end{align}
This completes the proof.
\end{proof}

\begin{fact}
The Pauli strings
$
    P_1 \otimes \cdots \otimes P_n
$
form a basis for $\C^{2^n \times 2^n}$. In particular, every Hermitian matrix is a real linear combination of Pauli strings.
\end{fact}

\begin{fact}
\label{fact:pauli-basis-su}
An operator $A$ lies in $\mathfrak{su}(2^n)$ if and only if there exist real numbers $\alpha_P \in \R$ such that
\begin{align}
    A
    =
    -i
    \sum_{P \in \cP_n^\circ}
    \alpha_P P,
\end{align}
where
$
    \cP_n^\circ
    :=
    \{ P_1 \otimes \cdots \otimes P_n
    \mid
    P_j \in \{I,X,Y,Z\}
    \}
    \setminus \{I^{\otimes n}\}.
$
\end{fact}

\begin{proof}
First the forward direction. Since each Pauli $P$ is Hermitian, the operator
\begin{align}
    H
    :=
    \sum_{P \in \cP_n^\circ}
    \alpha_P P
\end{align}
is Hermitian. Thus $A=-iH$ is skew-Hermitian. Also, every non-identity Pauli has trace $0$, so
\begin{align}
    \tr(A)
    =
    -i
    \sum_{P \in \cP_n^\circ}
    \alpha_P \tr(P)
    =
    0.
\end{align}
Thus $A \in \mathfrak{su}(2^n)$. Conversely, suppose $A \in \mathfrak{su}(2^n)$. Since $A$ is skew-Hermitian, $H:=iA$ is Hermitian. Every Hermitian matrix can be written as a real linear combination of Pauli strings, so
\begin{align}
    H
    =
    \beta_I I
    +
    \sum_{P \in \cP_n^\circ}
    \beta_P P
\end{align}
for some $\beta_I,\beta_P \in \R$. Taking traces gives
\begin{align}
    0
    =
    \tr(H)
    =
    \beta_I \tr(I)
    +
    \sum_{P \in \cP_n^\circ}
    \beta_P \tr(P)
    =
    2^n \beta_I.
\end{align}
Hence $\beta_I=0$, so
\begin{align}
    H
    =
    \sum_{P \in \cP_n^\circ}
    \beta_P P.
\end{align}
Since $A=-iH$, we get
\begin{align}
    A
    =
    -i
    \sum_{P \in \cP_n^\circ}
    \beta_P P
\end{align}
as desired.
\end{proof}

\begin{theorem}[Baker-Campbell-Hausdorff formula]
\label{thm:bch}
For $A,B \in \mathfrak{su}(2^n)$,
\begin{align}
    e^A e^B
    =
    \exp\left(
    A+B+\frac{1}{2}[A,B]
    +\frac{1}{12}[A,[A,B]]
    +\frac{1}{12}[B,[B,A]]
    +\cdots
    \right),
\end{align}
where $[A,B]=AB-BA$ is the \term{commutator} \cite{Hall2015LieGroups}.
\end{theorem}

\begin{theorem}[Trotterization]
\label{thm:trotter}
From \cite{Trotter1959ProductSemigroups}. For $A,B \in \mathfrak{su}(2^n)$,
\begin{align}
    e^{A+B}
    =
    \lim_{k \to \infty}
    \left(
    e^{A/k} e^{B/k}
    \right)^k.
\end{align}
\end{theorem}

\begin{definition}
\label{def:available-infinitesimal-generator}
Let $\mathcal{S} \subseteq SU(2^n)$ be a gate set, and let
$
    \mathcal{G}
    :=
    \overline{\langle \mathcal{S} \rangle}
$
be the closed group generated by $\mathcal{S}$. We say that $A \in \mathfrak{su}(2^n)$ is an \term{available infinitesimal generator} if
$
    e^{tA} \in \mathcal{G}
$
for every $t \in \mathbb{R}$.
\end{definition}

\begin{fact}[Generating sums]
\label{fact:generate-sums}
If $A$ and $B$ are available infinitesimal generators, then $A+B$ is also available as an infinitesimal generator.
\end{fact}

\begin{proof}
If $e^{tA}$ and $e^{tB}$ are available for all $t \in \R$, then by Theorem~\ref{thm:trotter},
\begin{align}
    e^{t(A+B)}
    =
    \lim_{k \to \infty}
    \left(
    e^{tA/k} e^{tB/k}
    \right)^k.
\end{align}
Thus $e^{t(A+B)}$ lies in the closure of the group generated by $e^{tA}$ and $e^{tB}$.
\end{proof}

\begin{fact}[Generating commutators]
\label{fact:generate-commutators}
If $A$ and $B$ are available infinitesimal generators, then $[A,B]$ is also available as an infinitesimal generator.
\end{fact}

\begin{proof}
Using the BCH formula,
\begin{align}
    e^{\epsilon A}
    e^{\epsilon B}
    e^{-\epsilon A}
    e^{-\epsilon B}
    =
    e^{\epsilon^2[A,B]+O(\epsilon^3)}.
\end{align}
Set $\epsilon=t/k^{1/2}$. Then
\begin{align}
    \left(
    e^{\epsilon A}
    e^{\epsilon B}
    e^{-\epsilon A}
    e^{-\epsilon B}
    \right)^k
    =
    \left(
    e^{(t^2/k)[A,B]+O(k^{-3/2})}
    \right)^k
    \longrightarrow
    e^{t^2[A,B]}.
\end{align}
Since $t \in \R$ is arbitrary, this completes the proof.
\end{proof}

\begin{definition}
\label{def:lie-algebra-generated-by-gates}
Let $\mathcal{S} \subseteq SU(2^n)$ be a gate set and let $\overline{\langle \mathcal{S}\rangle}$ be the closure of the group generated by $\mathcal{S}$. The \term{Lie algebra generated by $\mathcal{S}$} is the Lie algebra of the closed subgroup $\overline{\langle \mathcal{S}\rangle}$:
\begin{align}
    \Lie(\mathcal{S})
    :=
    \{ A \in \mathfrak{su}(2^n)
    \mid
    e^{tA} \in \overline{\langle \mathcal{S}\rangle}
    \text{ for all } t \in \R
    \}.
\end{align}
\end{definition}

\begin{theorem}
\label{thm:lie-algebra-universality}
Let $\mathcal{S} \subseteq SU(2^n)$. If
\begin{align}
    \Lie(\mathcal{S})
    =
    \mathfrak{su}(2^n),
\end{align}
then $\mathcal{S}$ is approximately universal on $n$ qubits, meaning
\begin{align}
    \overline{\langle \mathcal{S}\rangle}
    =
    SU(2^n).
\end{align}
\end{theorem}

\begin{proof}
By the closed subgroup theorem, $G=\overline{\langle \mathcal S\rangle}$
is a matrix Lie subgroup of $SU(2^n)$.
Since connected Lie subgroups are determined by their Lie algebras
\cite[Theorem~5.20]{Hall2015LieGroups}, and since $SU(2^n)$ is connected,
a closed subgroup of $SU(2^n)$ with Lie algebra $\mathfrak{su}(2^n)$ must be
all of $SU(2^n)$.
\end{proof}

\begin{definition}[Comparability-grid gadget]
\label{def:comp-grid-gadget}
Let $\CGrid$ be an ambient comparability grid. An \term{$n$-input, $n$-output gadget induced by $\CGrid$} is a tuple
\begin{align}
    \Gamma
    =
    (H,I,O,\mu),
\end{align}
where $H$ is an induced finite subgraph of $\CGrid$, $I=(i_1,\ldots,i_n)$ and $O=(o_1,\ldots,o_n)$ are ordered subsets of $V(H)$ with $|I|=|O|=n$, and $\mu$ assigns a measurement plane and angle parameter to each vertex in $V(H)\setminus O$. Operationally, $\Gamma$ defines a family of linear maps
\begin{align}
    G_\Gamma(\theta)
    :
    (\C^2)^{\otimes n}
    \to
    (\C^2)^{\otimes n},
\end{align}
where $\theta=(\theta_1,\ldots,\theta_m)$ are the free measurement angles. The map is obtained by placing the input state on $I$, preparing all vertices in $V(H)\setminus I$ in $\ket{+}$, applying $CZ$ along every edge of $H$, measuring the non-output vertices according to $\mu$, and applying the prescribed Pauli corrections. When this corrected map is unitary for all $\theta$, we call $\Gamma$ a \term{unitary gadget}.
\end{definition}

\begin{definition}[Special-unitary representative of a gadget]
\label{def:special-unitary-gadget}
Let $\Gamma$ be a unitary $n$-input, $n$-output gadget. Since global phases are irrelevant, we define its special-unitary representative by
\begin{align}
    \widetilde{G}_\Gamma(\theta)
    :=
    \det(G_\Gamma(\theta))^{-1/2^n} G_\Gamma(\theta)
    \in
    SU(2^n).
\end{align}
\end{definition}

\begin{definition}[Gate group and gadget Lie algebra]
\label{def:gadget-lie-algebra}
Let $\Gamma$ be a unitary gadget. Define the closed subgroup generated by the gadget as
\begin{align}
    \mathcal{G}_\Gamma
    :=
    \overline{
    \left\langle
    \widetilde{G}_\Gamma(\theta)
    \mid
    \theta \in \R^m
    \right\rangle
    }
    \subseteq
    SU(2^n).
\end{align}
Its Lie algebra is
\begin{align}
    \mathfrak{g}_\Gamma
    :=
    \Lie(\mathcal{G}_\Gamma)
    \subseteq
    \mathfrak{su}(2^n).
\end{align}
\end{definition}

\begin{lemma}[Zero-angle Clifford]
\label{lemma:zero-angle-clifford}
Let $\Gamma$ be a unitary gadget whose measurements are all in the $XY$ plane. Then the zero-angle gadget $\widetilde{G}_\Gamma(0,\ldots,0)$ is a Clifford operation. In particular,
\begin{align}
    \widetilde{G}_\Gamma(0,\ldots,0)
    \in
    \mathcal{G}_\Gamma.
\end{align}
\end{lemma}

\begin{proof}
At angle $0$, an $XY$-plane measurement is an $X$-basis measurement. The operations used to implement the gadget are preparation of $\ket{+}$ states, $CZ$ gates, Pauli measurements, and Pauli corrections. These are stabilizer operations. Since the corrected gadget is assumed to implement a unitary operation from $n$ inputs to $n$ outputs, the induced unitary must normalize the Pauli group, and hence is Clifford by Definition~\ref{def:clifford-group}. The final claim follows because $\widetilde{G}_\Gamma(0,\ldots,0)$ is one of the generators used in Definition~\ref{def:gadget-lie-algebra}.
\end{proof}

\begin{lemma}[Clifford inverses are available]
\label{lemma:clifford-inverses-available}
If $C \in \mathcal{G}_\Gamma$ is Clifford, then $C^\dagger \in \mathcal{G}_\Gamma$.
\end{lemma}

\begin{proof}
Modulo global phase, the Clifford group is finite. Hence $C$ has finite order: there exists $r \geq 1$ such that
$
    C^r = I.
$
Therefore
$
    C^\dagger
    =
    C^{-1}
    =
    C^{r-1}
    \in
    \mathcal{G}_\Gamma.
$
\end{proof}

\begin{lemma}[Conjugation by available Clifford gates]
\label{lemma:conjugation-available}
Let $A \in \mathfrak{g}_\Gamma$ and let $C \in \mathcal{G}_\Gamma$ be Clifford with $C^\dagger \in \mathcal{G}_\Gamma$. Then
$
    CAC^\dagger
    \in
    \mathfrak{g}_\Gamma.
$
\end{lemma}

\begin{proof}
Since $A \in \mathfrak{g}_\Gamma$, we have $e^{tA} \in \mathcal{G}_\Gamma$ for all $t \in \R$. Since $C,C^\dagger \in \mathcal{G}_\Gamma$,
$
    C e^{tA} C^\dagger
    \in
    \mathcal{G}_\Gamma.
$
But
$
    C e^{tA} C^\dagger
    =
    e^{tCAC^\dagger}.
$
Thus $e^{tCAC^\dagger} \in \mathcal{G}_\Gamma$ for all $t \in \R$, so $CAC^\dagger \in \mathfrak{g}_\Gamma$.
\end{proof}

\begin{fact}[Allowed Lie-algebra operations]
\label{fact:allowed-lie-operations}
Inside $\mathfrak{g}_\Gamma$, we may use the following closure operations:
\begin{align}
    A,B \in \mathfrak{g}_\Gamma
    &\implies
    A+B \in \mathfrak{g}_\Gamma, \\
    A,B \in \mathfrak{g}_\Gamma
    &\implies
    [A,B] \in \mathfrak{g}_\Gamma, \\
    A \in \mathfrak{g}_\Gamma,\; C \in \mathcal{G}_\Gamma
    &\implies
    CAC^\dagger \in \mathfrak{g}_\Gamma.
\end{align}
\end{fact}

\begin{proof}
The first implication is Fact~\ref{fact:generate-sums}. The second implication is Fact~\ref{fact:generate-commutators}. The third implication is Lemma~\ref{lemma:conjugation-available}.
\end{proof}

\begin{fact}[Initial $Z$-rotations]
\label{fact:initial-z-rotations}
Suppose the gadget $\Gamma$ has angle parameters allowing the implementation
\begin{align}
    \widetilde{G}_\Gamma(2t,0,\ldots,0)
    =
    C e^{-itZ_1}
\end{align}
for some Clifford $C \in \mathcal{G}_\Gamma$ and all $t \in \R$. Then
$
    -iZ_1
    \in
    \mathfrak{g}_\Gamma.
$
More generally, if for each $j \in [n]$ there is a Clifford $C_j \in \mathcal{G}_\Gamma$ such that
\begin{align}
    \widetilde{G}_\Gamma(0,\ldots,2t,\ldots,0)
    =
    C_j e^{-itZ_j},
\end{align}
then
$
    -itZ_j
    \in
    \mathfrak{g}_\Gamma
$
for all $t \in \R$ and all $j \in [n]$.
\end{fact}

\begin{proof}
By Lemma~\ref{lemma:clifford-inverses-available}, $C^\dagger \in \mathcal{G}_\Gamma$. Therefore
\begin{align}
    C^\dagger \widetilde{G}_\Gamma(2t,0,\ldots,0)
    =
    e^{-itZ_1}
    \in
    \mathcal{G}_\Gamma
\end{align}
for all $t \in \R$. Hence $-iZ_1 \in \mathfrak{g}_\Gamma$. The same argument applies to each $j$.
\end{proof}

\begin{theorem}[Gadget universality via Pauli generation]
\label{thm:gadget-universality-pauli-generation}
Let $\Gamma$ be a unitary $n$-input, $n$-output gadget. If
\begin{align}
    -iP \in \mathfrak{g}_\Gamma
\end{align}
for every non-identity tensor-product Pauli $P \in \cP_n^\circ$, then $\Gamma$ is approximately universal on $n$ qubits.
\end{theorem}

\begin{proof}
By Fact~\ref{fact:pauli-basis-su}, the elements $-iP$ for non-identity Pauli strings $P$ span $\mathfrak{su}(2^n)$ as a real vector space. Since $\mathfrak{g}_\Gamma$ is closed under real linear combinations by Fact~\ref{fact:generate-sums},
$
    \mathfrak{g}_\Gamma
    =
    \mathfrak{su}(2^n).
$
Therefore $\mathcal{G}_\Gamma=SU(2^n)$ by Theorem~\ref{thm:lie-algebra-universality}, so the gadget is approximately universal.
\end{proof}

\begin{example}
Consider the diagonal gadget. The relevant Clifford operation is
\begin{align}
    C
    =
    \CNOT_{2,0}
    \CZ_{0,1}
    (H_0 H_1 H_2),
\end{align}
where gates are written in the order in which they are applied. Starting from
\begin{align}
    ZII,
    \qquad
    IZI,
    \qquad
    IIZ,
\end{align}
we can generate new operators using $P \mapsto CPC^\dagger$ and closure. The following pseudocode implements exactly this search.

\begin{algorithm}[H]
\caption{$3$-qubit approximate universality check for the diagonal gadget}
\label{alg:three-qubit-gadget-universality}
\begin{algorithmic}[1]
\State $C \gets \CNOT_{2,0}\CZ_{0,1}(H_0H_1H_2)$
\State $\mathrm{Queue} \gets [ZII, IZI, IIZ]$
\State $\mathrm{Found} \gets \varnothing$

\While{$\mathrm{Queue}$ is nonempty}
    \State Pop a Pauli $P$ from $\mathrm{Queue}$

    \If{$P \notin \mathrm{Found}$}
        \State Add $CPC^\dagger$ to $\mathrm{Queue}$

        \For{$R \in \mathrm{Found}$}
            \If{$P$ and $R$ anticommute}
                \State Add $PR$, up to phase, to $\mathrm{Queue}$
            \EndIf
        \EndFor

        \State Add $P$ to $\mathrm{Found}$
    \EndIf
\EndWhile

\If{$\mathrm{Found}$ contains all $63$ non-identity three-qubit Paulis}
    \State Conclude $\mathfrak{g}=\mathfrak{su}(8)$
    \State Conclude the gadget is approximately universal
\EndIf
\end{algorithmic}
\end{algorithm}
It turns out that the diagonal gadget fails to generate all 63 non-identity Pauli strings. On the other hand, running the above algorithm for the width-3 gadget succeeds.
\end{example}
